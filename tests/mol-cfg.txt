
>>> from umling.mol import *
>>> from umling.mol.symbols import (
...      S, NP, VP, Det, N, V, this, these, dog, dogs, barks, bark, sg, pl
... )
...
>>> from umling.mol.variables import form

>>> R(S, NP[form], VP[form])
>>> R(NP[form], Det[form], N[form])
>>> R(Det[sg], this)
>>> R(Det[pl], these)
>>> R(N[sg], dog)
>>> R(N[pl], dogs)
>>> R(VP[form], V[form])
>>> R(V[sg], barks)
>>> R(V[pl], bark)

>>> grammar = done()
>>> len(grammar.rules)
9

>>> print(grammar)
S[] -> NP[form] VP[form]
NP[form] -> Det[form] N[form]
Det[sg] -> this
Det[pl] -> these
N[sg] -> dog
N[pl] -> dogs
VP[form] -> V[form]

V[pl] -> bark

>>> sorted(grammar.terminals())
[bark, barks, dog, dogs, these, this]

>>> sorted(grammar.first(S), key=repr)
[Det[U], NP[U], these, this]

>>> dr = grammar.continuations(this)[0]
>>> dr
Det[sg] -> this
>>> d = dr.shift(0)
>>> d
Det(0:1)
>>> print(d)
Det[sg]
  this

>>> r = grammar.continuations(d.cat)[0]
>>> r
NP[form] -> Det[form] N[form]
>>> m1 = r.start()
>>> m1
NP[form] -> * Det[form] N[form]
>>> r.start() * d
NP[form] -> Det[sg] * N[form]
>>> m2 = m1 * d
>>> m2
NP[form] -> Det[sg] * N[form]
>>> m2.bindings
{form: sg}
>>> e = m2.expectation()
>>> e
N[sg]
>>> words = sorted(grammar.first_terminals(e))
>>> words
[dog]
>>> nr = grammar.continuations(words[0])[0]
>>> nr
N[sg] -> dog
>>> n = nr.shift(1)
>>> n
N(1:2)
>>> m3 = m2 * n
>>> m3
NP[form] -> Det[sg] N[sg] *
>>> m3.is_complete()
True
>>> m3.children
[Det(0:1), N(1:2)]

>>> np = m3.reduce()
>>> print(np)
NP[sg]
  Det[sg]
    this
  N[sg]
    dog

>>> sorted(grammar.first(S), key=repr)
[Det[U], NP[U], these, this]

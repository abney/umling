## unify

>>> (isg+pl) & (sg+pl)
pl
>>> isg & (sg+pl)
∅
>>> bool(isg & (sg+pl))
False
>>> bool(sg)
True

>>> from umling.mol.variables import form
>>> bindings = {}
>>> N[form].unify(N[sg+pl], bindings)
N[pl+sg]
>>> bindings
{form: pl+sg}
>>> bindings = {}
>>> N[isg+pl].unify(N[sg+pl], bindings)
N[pl]
>>> bindings
{}
>>> bindings = {form: isg+pl}
>>> N[form].unify(N[sg+pl], bindings)

>>> bindings
{form: pl}


## 1.
>>> from umling.mol import *
>>> from umling.mol.symbols import a, b, c, d
>>> E(1, a, 2)
>>> E(2, b, 2)
>>> F(2)
>>> fsa = done()

## 2.
>>> print(fsa)
E(1, a, 2)
E(2, b, 2)
F(2)
>>> enum(fsa, 3)
[0] (a,)
[1] (a, b)
[2] (a, b, b)
...
>>> a * b * b in fsa
True
>>> epsilon in fsa
False
>>> fsa(a * b)
True

## 3.
>>> io(a,b)
a:b
>>> R = io(a,epsilon) * star(io(a,b))
>>> R(a * a * a)
[0] (b, b)
>>> R([a, a, a])
[0] (b, b)
>>> R2 = io(a * b * c, c * (a + b))
>>> R2
(a∙b∙c):(c∙(a+b))
>>> sorted(R2(a * b * c))
[(c, a), (c, b)]

## 4.

## 5.
>>> T2 = T @ star(io(a,c) + io(b,d))
>>> T2
(<FST with 3 states>@(a:c + b:d)*)
>>> T2([a, a, b, a, b])
[0] (c, c, d)
>>> T3 = T2 * io(epsilon, a)
>>> T3([a, a, b, a, b])
[0] (c, c, d, a)

## 6.
>>> from umling.mol.symbols import (
...     cat, tease, dog, s,
...     k, ae, t, i, z, d, o, g
... )
...
>>> lex1 = (
...     io(cat, k*ae*t) +
...     io(tease, t*i*z) +
...     io(dog, d*o*g) +
...     io(s,z)
... )
...
>>> lex1(dog)
[0] (d, o, g)
>>> lex = repeat(lex1)
>>> lex([cat, s, tease, dog, s])
[0] (k, ae, t, z, t, i, z, d, o, g, z)

## 8.
>>> rule = rewrite(z, s, after=t)
>>> rule
z → s / t _ ε
>>> T2 = lex @ rule

[0] (k, ae, t, s, t, i, z, d, o, g, z)
>>> invert(T2)([k, ae, t, s, t, i, z, d, o, g, z])
